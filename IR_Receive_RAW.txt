// Define the pin where your IR Receiver is connected
const int IR_RECEIVE_PIN = 15; 

void setup() {
    Serial.begin(115200);

    Serial.println("----------------------------------------");
    Serial.println("READY TO DECODE SIGNALS");
    Serial.println("Point your remote and press a button...");

}

void loop() {
    if (IrReceiver.decode()) {
        
        // 1. Print the basic summary (Protocol, Address, Command, etc.)
        // This gives you the "PulseDistance" or "NEC" line you saw in your screenshots.
        IrReceiver.printIRResultShort(&Serial);

        // 2. Print the Raw Timing Array
        // This is CRITICAL for "PulseDistance" or "Unknown" signals.
        // It prints the "uint16_t rawIRTimings[] = {...}" block automatically.
        IrReceiver.printIRSendUsage(&Serial);

        // 3. Custom Logic to Separate Protocols
        if (IrReceiver.decodedIRData.protocol == NEC) {
            Serial.println(">>> TYPE: Standard NEC detected.");
            Serial.print(">>> Use this to send: IrSender.sendNEC(0x");
            Serial.print(IrReceiver.decodedIRData.address, HEX);
            Serial.print(", 0x");
            Serial.print(IrReceiver.decodedIRData.command, HEX);
            Serial.println(", 0);");
        } 
        else if (IrReceiver.decodedIRData.protocol == PULSE_DISTANCE) {
            Serial.println(">>> TYPE: PulseDistance (Generic) detected.");
            Serial.println(">>> DATA: Copy the 'rawIRTimings' array printed above to send this.");
            Serial.print(">>> RAW HEX (for logic): 0x");
            Serial.println(IrReceiver.decodedIRData.decodedRawData, HEX);
        }
        else if (IrReceiver.decodedIRData.protocol == UNKNOWN) {
            Serial.println(">>> TYPE: Unknown Protocol.");
            Serial.println(">>> ACTION: You MUST use the 'rawIRTimings' array printed above.");
        }

        Serial.println("----------------------------------------");
        
        // Resume listening for the next signal
        IrReceiver.resume(); 
    }
}
